// Lógica para exportación a CSV function exportCSV() { // Asegúrate de que esta función acceda a tu fuente de datos actual (allRows) const rows = window.allRows ? window.allRows() : []; const hdr = 'Fecha,Dia,Vol,VolProy,EvAM,AusAM,Presentes,EvPM,Rutas,GGPest,DS,Tipo'; const csvContent = [hdr, ...rows.map(r => `${String(r.d).padStart(2,'0')}/06/2026,${r.dnom},${r.vol||0},${r.volProy||0},${r.evAM||0},${r.ausAM||0},${r.presentes||0},${r.evPM||0},${r.rutas||0},${r.dot?r.dot.ggpEst:0},${r.ds?r.ds:0},${r.tipo||''}` )].join('\n'); const blob = new Blob([csvContent], { type: 'text/csv;charset=utf-8;' }); const link = document.createElement('a'); link.href = URL.createObjectURL(blob); link.download = 'datos_operativos.csv'; link.click(); } // Asegurar que al cargar la página se inicie la escucha window.addEventListener('firebase-ready', () => { window.fbStore.listen(data => { window.rawData = data; renderAll(); }); });
